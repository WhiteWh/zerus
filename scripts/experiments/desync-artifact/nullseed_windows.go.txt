// Zerus MIT: keep desync's null-seed scratch file process-owned on Windows.
package desync

import (
	"crypto/rand"
	"encoding/hex"
	"os"
	"path/filepath"

	"golang.org/x/sys/windows"
)

func init() {
	zerusCreateNullSeedTemp = func(dir, pattern string) (*os.File, error) {
		random := make([]byte, 16)
		if _, err := rand.Read(random); err != nil {
			return nil, err
		}
		path := filepath.Join(dir, pattern+hex.EncodeToString(random))
		name, err := windows.UTF16PtrFromString(path)
		if err != nil {
			return nil, err
		}
		h, err := windows.CreateFile(name, windows.GENERIC_READ|windows.GENERIC_WRITE|windows.DELETE,
			windows.FILE_SHARE_READ|windows.FILE_SHARE_WRITE|windows.FILE_SHARE_DELETE,
			nil, windows.CREATE_NEW,
			windows.FILE_ATTRIBUTE_TEMPORARY|windows.FILE_FLAG_DELETE_ON_CLOSE, 0)
		if err != nil {
			return nil, err
		}
		return os.NewFile(uintptr(h), path), nil
	}
}
